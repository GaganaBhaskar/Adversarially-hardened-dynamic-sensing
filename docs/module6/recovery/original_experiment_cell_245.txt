def tensor_hash(tensor):

    array = (
        tensor.detach()
        .cpu()
        .contiguous()
        .numpy()
    )

    digest = hashlib.sha256()
    digest.update(str(array.dtype).encode())
    digest.update(str(array.shape).encode())
    digest.update(array.tobytes(order="C"))

    return digest.hexdigest()


def model_hashes():
    return {
        name: tensor_hash(value)
        for name, value in MODEL.state_dict().items()
    }

def event_active(scenario, frame):

    return any(
        start <= frame < end
        for start, end in scenario["windows"]
    )


def frame_phase(scenario, frame):

    if not scenario["windows"]:
        return "clean"

    if event_active(scenario, frame):
        return "event"

    if frame < scenario["windows"][0][0]:
        return "pre_event"

    return "recovery"


def optical_frame(clean, scenario, frame):

    if not event_active(scenario, frame):
        return clean

    kind = scenario["kind"]

    if kind == "laser":

        attacked, attack_map = apply_optical_attack(
            clean,
            ATTACK,
            time_step=frame,
        )

        return attacked

    if kind == "brightening":
        return clean * 1.5

    if kind == "dimming":
        return clean * 0.45

    if kind == "benign_local":

        height, width = clean.shape[-2:]

        y = torch.linspace(
            0, 1, height,
            dtype=clean.dtype,
            device=clean.device,
        )

        x = torch.linspace(
            0, 1, width,
            dtype=clean.dtype,
            device=clean.device,
        )

        yy, xx = torch.meshgrid(
            y, x,
            indexing="ij",
        )

        illumination = 2.0 * torch.exp(
            -(
                (xx - 0.5) ** 2
                + (yy - 0.5) ** 2
            ) / (2 * 0.12 ** 2)
        )

        return clean + illumination.unsqueeze(0)

    raise ValueError(kind)

    return simulate_sensor(

def confusion_matrix(prediction, target):

    encoded = (
        target.reshape(-1).long() * 4
        + prediction.reshape(-1).long()
    )

    return torch.bincount(
        encoded,
        minlength=16,
    ).reshape(4, 4).cpu().numpy()


def summarize_confusion(matrix):

    matrix = np.asarray(
        matrix,
        dtype=np.float64,
    )

    tp = np.diag(matrix)

    actual = matrix.sum(axis=1)

    predicted = matrix.sum(axis=0)

    union = actual + predicted - tp

    denominator = actual + predicted

    iou = [
        float(tp[i] / union[i])
        if union[i] > 0 else None
        for i in range(4)
    ]

    dice = [
        float(2 * tp[i] / denominator[i])
        if denominator[i] > 0 else None
        for i in range(4)
    ]

    foreground = [
        value for value in iou[1:]
        if value is not None
    ]

    foreground_dice = [
        value for value in dice[1:]
        if value is not None
    ]

    return {
        "pixel_accuracy": float(
            tp.sum() / matrix.sum()
        ),
        "foreground_miou": (
            float(np.mean(foreground))
            if foreground else None
        ),
        "foreground_mean_dice": (
            float(np.mean(foreground_dice))
            if foreground_dice else None
        ),
        "class_iou": dict(zip(CLASS_NAMES, iou)),
        "confusion_matrix": (
            matrix.astype(np.int64).tolist()
        ),
    }

def build_model_input(readout):

    rgb = readout["readout"]

    mask = readout["sampling_mask"]

    assert rgb.shape == (3, 256, 256)

    assert mask.shape == (256, 256)

    assert (rgb[:, ~mask.bool()] == 0).all()

    return torch.cat(
        [
            rgb,
            mask.float().unsqueeze(0),
        ],
        dim=0,
    )


def predict_batch(inputs):

    batch = torch.stack(inputs).to(DEVICE)

    logits = MODEL(batch)

    assert logits.shape == (
        len(inputs), 4, 256, 256
    )

    assert torch.isfinite(logits).all()

    return logits.argmax(dim=1).cpu()

# Cache identical perception inputs.
# This is an execution optimization only.
# Controllers and sensors still execute every frame.

from collections import OrderedDict

PREDICTION_CACHE = OrderedDict()

CACHE_LIMIT = 128

ACTUAL_MODEL_INFERENCES = 0

CACHE_HITS = 0


def cached_predictions(inputs):

    global ACTUAL_MODEL_INFERENCES
    global CACHE_HITS

    results = [None] * len(inputs)

    missing = {}

    for i, value in enumerate(inputs):

        key = tensor_hash(value)

        if key in PREDICTION_CACHE:

            results[i] = PREDICTION_CACHE[key]

            PREDICTION_CACHE.move_to_end(key)

            CACHE_HITS += 1

        else:

            if key not in missing:
                missing[key] = {
                    "input": value,
                    "indices": [],
                }

            else:
                CACHE_HITS += 1

            missing[key]["indices"].append(i)

    if missing:

        keys = list(missing)

        predictions = predict_batch([
            missing[key]["input"]
            for key in keys
        ])

        ACTUAL_MODEL_INFERENCES += len(keys)

        for key, prediction in zip(
            keys, predictions
        ):

            PREDICTION_CACHE[key] = prediction

            for index in missing[key]["indices"]:
                results[index] = prediction

        while len(PREDICTION_CACHE) > CACHE_LIMIT:
            PREDICTION_CACHE.popitem(last=False)

    assert all(value is not None for value in results)

    return results
